\section{Background}
%
\begin{frame}{Background: Project Emerge}
    \begin{columns}
        \column{0.5\textwidth}
        \centering
        \textit{Project Emerge}: dimostratore open-source che porta l'\textbf{Aggregate Computing} dall'ambito teorico a un'implementazione reale, applicata per il coordinamento di sciami robotici

        \column{0.5\textwidth}
        \begin{block}{Aggregate Computing}
            Paradigma per programmare sciami di dispositivi eterogenei come un'unica entità collettiva -- senza controllo centralizzato
        \end{block}

        \vspace{1.4em}

        \begin{block}{Architettura}
            4 moduli -- ognuno responsabile di una determinata mansione -- e robot comunicanti tramite un broker \textbf{MQTT} centralizzato
        \end{block}
    \end{columns}
\end{frame}
%

\section{Analisi}
%
\begin{frame}{Analisi: obiettivi, vincoli e requisiti}
    \centering
    Obiettivo: rendere il sistema \textbf{protocol-agnostic}, rendendo possibile a client eterogenei di interagire con il sistema

    \vspace{1.5em}

    \begin{columns}
        \column{0.33\textwidth}
        \begin{block}{Vincoli}
            \begin{itemize}
                \item Non modificare i moduli esistenti
                \item MQTT resta il protocollo
            \end{itemize}
        \end{block}

        \column{0.33\textwidth}
        \begin{block}{Requisiti funzionali}
            Esporre più protocolli eterogenei, con supporto sia a request/response sia a notifica, ed estendibilità futura
        \end{block}

        \column{0.33\textwidth}
        \begin{block}{Requisiti non funzionali}
            Molte connessioni simultanee, basse latenze, estendibilità senza toccare il nucleo, testabilità
        \end{block}
    \end{columns}
\end{frame}
%

\section{Progettazione}
%
\begin{frame}{Progettazione: il \textit{domain gateway}}
    \begin{columns}
        \column{0.5\textwidth}
        \centering
        Un unico modulo aggiuntivo si frappone tra il broker MQTT e i client eterogenei, mediando la comunicazione tra i due

        \vspace{1em}

        \begin{block}{Vantaggi}
            \begin{itemize}
                \item Moduli esistenti invariati
                \item Nuovi protocolli $\Rightarrow$ si tocca solo il gateway
                \item Supporta sia notifica continua sia request/response (grazie a una \textbf{cache} interna)
            \end{itemize}
        \end{block}

        \column{0.5\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\linewidth]{figures/arcchiettura-generale.pdf}
            \caption{\centering Architettura del sistema con l'introduzione del gateway -- in rosso le aggiunte}

\end{frame}

\section{Valutazione}
%
\begin{frame}{Valutazione: testing automatico}
    \centering
    La Dependency injection rende possibile iniettare connessioni Mock e isolare facilmente ogni singolo componente, senza modificare il codice applicativo.

    \vspace{2em}

    \begin{columns}
        \column{0.48\textwidth}
        \begin{block}{Strumenti e struttura}
            \begin{itemize}
                \item Organizzazione per protocollo
                \item \texttt{MockMQTTConnection} iniettato al posto della reale connessione
            \end{itemize}
        \end{block}

        \column{0.48\textwidth}
        \begin{block}{Risultati}
            \begin{itemize}
                \item 80 test, tutti superati in $\sim$6s
                \item Copertura complessiva: \textbf{88\%}
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}
